export async function onRequest() {
  const rss = 'https://www.ynet.co.il/Integration/StoryRss1854.xml';

  try {
    const response = await fetch(rss, {
      headers: {
        'User-Agent': 'Mozilla/5.0',
        'Accept': 'application/rss+xml, application/xml, text/xml, */*'
      },
      cf: { cacheTtl: 60, cacheEverything: false }
    });

    if (!response.ok) {
      return new Response(JSON.stringify({ items: [], error: 'ynet fetch failed' }), {
        status: 502,
        headers: { 'content-type': 'application/json; charset=utf-8' }
      });
    }

    const xml = await response.text();
    const matches = [...xml.matchAll(/<item>[\s\S]*?<title>(?:<!\[CDATA\[)?([\s\S]*?)(?:\]\]>)?<\/title>[\s\S]*?<\/item>/gi)];

    const items = matches.slice(0, 15).map(m => ({
      title: m[1]
        .replace(/<!\[CDATA\[|\]\]>/g, '')
        .replace(/&quot;/g, '"')
        .replace(/&amp;/g, '&')
        .replace(/&#39;/g, "'")
        .replace(/&lt;/g, '<')
        .replace(/&gt;/g, '>')
        .trim()
    })).filter(x => x.title);

    return new Response(JSON.stringify({ items, updated: new Date().toISOString() }), {
      headers: {
        'content-type': 'application/json; charset=utf-8',
        'cache-control': 'no-store, max-age=0'
      }
    });
  } catch (e) {
    return new Response(JSON.stringify({ items: [], error: String(e) }), {
      status: 500,
      headers: { 'content-type': 'application/json; charset=utf-8' }
    });
  }
}
